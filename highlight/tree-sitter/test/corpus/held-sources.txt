==================
Held names a parameter, a source, a binding, a field and an end
==================

named: (held: list) -> (r: item from held) =
    r = held.val
end named
both: (held: list, other: list) -> (r: item from held, other) =
    held := 1
end both
twice: (held: list) -> (r: item from held held, s: u8) =
    r = held(held)
end twice
held: (a: list) -> none =
    a.held = 1
end held
