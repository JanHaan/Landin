link : (link : i32, option : i32, caller : i32) -> (as : i32) =
  as = g(option, link)
  as = g(link)
  as = (caller)
  record = (caller: link, other: option)
  made = pair(caller: 1, other: caller)
  option: loop do
    break option when link > as
  end option
end link
