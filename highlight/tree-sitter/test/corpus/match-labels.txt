f: (c: compass) -> (n: i32) =
    match c
        north, south: n = 1
        ptr (p): n = 2
        _: n = 3
    end match
end f

==================

f: (sink m: maybe) -> (n: i32) =
    match m
        absent: n = 1
        ptr (sink p): n = release(p)
    end match
end f
